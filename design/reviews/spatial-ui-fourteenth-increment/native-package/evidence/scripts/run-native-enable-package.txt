#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fourteenth
run=$base/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$base/frozen/xuzz" --backend opengl --resize 1440,1000 --profile
 --chord 'Ctrl+Alt+Shift+L' --key right --key enter --key enter
 --dump-a11y --capture "$run/raw/05-retained-package-status.ppm" --wait-ms 1500
 --key right --key enter
 --dump-a11y --capture "$run/raw/06-review-disabled-package.ppm" --wait-ms 1500
 --key right --key right --key right --key right --key right --key enter
 --dump-a11y --capture "$run/raw/07-review-enabled-package.ppm" --wait-ms 1500
 --key right --key right --key right --key right --key right --key right --key enter
 --dump-a11y --capture "$run/raw/08-selected-package-link.ppm" --wait-ms 1500
 --chord 'Alt+Shift+J' --chord 'Alt+Shift+J' --chord 'Alt+Shift+X'
 --chord 'Alt+Shift+J' --chord 'Alt+Shift+J' --chord 'Alt+Shift+J'
 --dump-a11y --capture "$run/raw/09-independent-package-cursors.ppm" --wait-ms 1500
 --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/enable-package-argv.txt"; printf '\n' >> "$run/evidence/enable-package-argv.txt"
LD_LIBRARY_PATH="$base/frozen${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/enable-package.log" 2>&1
