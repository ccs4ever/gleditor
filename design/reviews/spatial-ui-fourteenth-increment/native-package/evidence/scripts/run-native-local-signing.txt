#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fourteenth
run=$base/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$base/frozen/xuzz" --backend opengl --resize 1440,1000 --profile
 --chord 'Ctrl+Shift+S' --key tab --key tab --type 'Spatial Validation'
 --key tab --type 'spatial@example.invalid' --key enter --wait-ms 5000
 --dump-a11y --capture "$run/raw/02-local-publication-outcome.ppm" --wait-ms 2000
 --key escape --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/local-signing-argv.txt"; printf '\n' >> "$run/evidence/local-signing-argv.txt"
LD_LIBRARY_PATH="$base/frozen${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/local-signing.log" 2>&1
