#!/usr/bin/env bash
set -euo pipefail
run=/tmp/spatial-tenth-cancel.20261015/candidate
bin=$run/bin
store="$run/data/xudu/xanadocs/untitled-20261015-131855"
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --chord Ctrl+O --key tab --type "$store" --key enter --wait-ms 9000
  --chord Ctrl+H --wait-ms 9000 --dump-a11y
  --click-label 'T · Version b1' --wait-ms 9000
  --click-label 'Close hypertime' --wait-ms 5000
  --dump-a11y --capture "$run/raw/08-reopened-b1.ppm"
  --chord 'Alt+Shift+Return' --wait-ms 9000
  --dump-a11y --capture "$run/raw/09-retry-entered-b1.ppm"
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/reopen-exact-argv.txt"
printf '\n' >> "$run/evidence/reopen-exact-argv.txt"
sha256sum "$bin/xuzz" "$bin/libgleditor.so.0" "$bin/xudu-dump" > "$run/evidence/reopen-exact-hashes-before.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' \
  timeout --signal=TERM --kill-after=5s 120s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' \
  "${app[@]}" > "$run/evidence/reopen-exact-session.log" 2>&1
sha256sum "$bin/xuzz" "$bin/libgleditor.so.0" "$bin/xudu-dump" > "$run/evidence/reopen-exact-hashes-after.txt"
for f in "$run"/raw/08-reopened-b1.ppm "$run"/raw/09-retry-entered-b1.ppm; do [[ -f $f ]] && sha256sum "$f"; done > "$run/evidence/reopen-exact-ppm-sha256.txt"
