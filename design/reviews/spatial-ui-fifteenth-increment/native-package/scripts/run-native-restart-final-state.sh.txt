#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fifteenth
run=$base/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$base/frozen-final/xuzz" --backend opengl --resize 1440,1000 --profile --wait-ms 1500
 --dump-a11y --capture "$run/raw/28-restarted-final-quiet-pin.ppm" --wait-ms 1500 --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/restart-final-state-argv.txt"; printf '\n' >> "$run/evidence/restart-final-state-argv.txt"
LD_LIBRARY_PATH="$base/frozen-final${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 120s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/restart-final-state.log" 2>&1
target="$run/evidence/checkpoints/28-restarted-final-quiet-pin"
mkdir -p "$target"
cp "$run/data/xudu/package-visibility" "$target/package-visibility.bencode.txt"
sha256sum "$run/data/xudu/package-visibility" > "$target/package-visibility.sha256"
mapfile -t stores < <(find "$run/data/xudu/xanadocs" -mindepth 1 -maxdepth 1 -type d ! -name default | sort)
for index in "${!stores[@]}"; do
 "$base/frozen-final/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$index]}" > "$target/source-$index.ops.txt"
 sha256sum "${stores[$index]}/ops.nodes" "${stores[$index]}/store.tables" > "$target/source-$index.files.sha256"
done
"$base/frozen-final/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt"
