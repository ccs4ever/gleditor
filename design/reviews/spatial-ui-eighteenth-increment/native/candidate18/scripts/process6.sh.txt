#!/usr/bin/env bash
set -euo pipefail
repo=$PWD
run=$repo/build/validation-eighteenth/native/candidate18
app=$repo/build/validation-eighteenth/frozen/xuzz
libdir=$repo/build/validation-eighteenth/frozen
export SDL_VIDEODRIVER=x11 SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg PATH=/tmp/spatial-xvfb/usr/bin:$PATH
args=(--backend opengl --resize 1440,1000 --profile --wait-ms 5000
  --chord Ctrl+O --wait-ms 700 --key enter
  --key down --key down --key down --key down --key down
  --wait-ms 200 --dump-a11y --capture "$run/raw/19-open-list-download-row-visible.ppm"
  --key escape --key escape --chord Ctrl+Q)
printf '%q ' "$app" "${args[@]}" > "$run/argv-process6.txt"
printf '\n' >> "$run/argv-process6.txt"
set +e
LD_LIBRARY_PATH="$libdir${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" timeout --signal=TERM --kill-after=5s 300s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' dbus-run-session -- "$app" "${args[@]}" > "$run/logs/process6.log" 2>&1
code=$?
set -e
printf '%s\n' "$code" > "$run/process6.exit"
exit "$code"
