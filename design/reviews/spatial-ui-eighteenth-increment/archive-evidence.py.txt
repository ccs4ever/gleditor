#!/usr/bin/env python3
"""Archive eighteenth-increment strict renders and validation attempts.

Run only after the owning native evidence workflow is complete, with
--native-complete. This helper deliberately does not read or copy native
profiles, keys, stores, payloads, or frames.
"""

from __future__ import annotations

import argparse
import gzip
import hashlib
import json
import shutil
import struct
import subprocess
import zlib
from pathlib import Path


ROOT = Path(__file__).resolve().parents[2]
STAGING = ROOT / "build/validation-eighteenth"
STRICT = STAGING / "strict-core"
REVIEW = ROOT / "design/reviews/spatial-ui-eighteenth-increment"
STRICT_OUT = REVIEW / "strict-core"
ATTEMPTS_OUT = REVIEW / "attempts"


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def gzip_bytes(data: bytes) -> bytes:
    import io

    raw = io.BytesIO()
    with gzip.GzipFile(filename="", mode="wb", fileobj=raw, mtime=0) as out:
        out.write(data)
    return raw.getvalue()


def png_chunk(kind: bytes, payload: bytes) -> bytes:
    body = kind + payload
    return struct.pack(">I", len(payload)) + body + struct.pack(">I", zlib.crc32(body))


def write_png(path: Path, width: int, height: int, rgb: bytes) -> None:
    rows = b"".join(
        b"\0" + rgb[row * width * 3 : (row + 1) * width * 3]
        for row in range(height)
    )
    header = struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0)
    payload = zlib.compress(rows, level=9)
    path.write_bytes(
        b"\x89PNG\r\n\x1a\n"
        + png_chunk(b"IHDR", header)
        + png_chunk(b"IDAT", payload)
        + png_chunk(b"IEND", b"")
    )


def verify_png(path: Path, width: int, height: int, rgb: bytes) -> None:
    data = path.read_bytes()
    if not data.startswith(b"\x89PNG\r\n\x1a\n"):
        raise RuntimeError(f"bad PNG signature: {path}")
    offset = 8
    compressed = bytearray()
    saw_header = saw_end = False
    while offset < len(data):
        length = struct.unpack_from(">I", data, offset)[0]
        offset += 4
        kind = data[offset : offset + 4]
        offset += 4
        body = data[offset : offset + length]
        offset += length
        crc = struct.unpack_from(">I", data, offset)[0]
        offset += 4
        if zlib.crc32(kind + body) != crc:
            raise RuntimeError(f"PNG CRC mismatch: {path}")
        if kind == b"IHDR":
            if struct.unpack(">IIBBBBB", body) != (width, height, 8, 2, 0, 0, 0):
                raise RuntimeError(f"unexpected PNG dimensions or format: {path}")
            saw_header = True
        elif kind == b"IDAT":
            compressed.extend(body)
        elif kind == b"IEND":
            saw_end = True
            break
    if not saw_header or not saw_end:
        raise RuntimeError(f"incomplete PNG: {path}")
    decoded = zlib.decompress(compressed)
    row_bytes = width * 3
    if len(decoded) != height * (row_bytes + 1):
        raise RuntimeError(f"wrong decoded raster length: {path}")
    for row in range(height):
        start = row * (row_bytes + 1)
        if decoded[start] != 0 or decoded[start + 1 : start + row_bytes + 1] != rgb[
            row * row_bytes : (row + 1) * row_bytes
        ]:
            raise RuntimeError(f"PNG RGB differs from PPM at row {row}: {path}")


def read_ppm(path: Path) -> tuple[int, int, bytes]:
    data = path.read_bytes()
    # Renderer PPMs use the canonical single-line P6 header. Reject other
    # shapes rather than guessing where the binary raster begins.
    prefix = data.split(b"\n", 3)
    if len(prefix) != 4 or prefix[0] != b"P6":
        raise RuntimeError(f"unexpected PPM header: {path}")
    try:
        width, height = (int(value) for value in prefix[1].split())
        maximum = int(prefix[2])
    except ValueError as error:
        raise RuntimeError(f"invalid PPM dimensions: {path}") from error
    if maximum != 255:
        raise RuntimeError(f"unexpected PPM range: {path}")
    rgb = prefix[3]
    if len(rgb) != width * height * 3:
        raise RuntimeError(f"unexpected PPM raster size: {path}")
    return width, height, rgb


def require_exit(name: str, expected: int) -> Path:
    path = Path("/tmp") / f"spatial-eighteenth-{name}.exit"
    if not path.is_file():
        raise RuntimeError(f"missing exit code: {path}")
    got = path.read_text(encoding="utf-8").strip()
    if got != str(expected):
        raise RuntimeError(f"{name} exit was {got!r}, expected {expected}")
    return path


def archive_logs() -> dict[str, dict[str, object]]:
    attempts: dict[str, dict[str, object]] = {}
    run_specs = {
        "canonical-build": ("build", 0),
        "canonical-build-after-matrix": ("build-final", 0),
        "ui-final": ("ui", 0), "xuzz-final": ("xuzz", 0),
        "owning-publication-packages": ("owning", 0),
        "strict-core-backends": ("strict", 0),
        "format": ("format", 0), "lint": ("lint", 0),
        "text-policy": ("text-policy", 0),
    }
    for label, (run, expected) in run_specs.items():
        log_path = Path("/tmp") / f"spatial-eighteenth-{run}.log"
        if not log_path.is_file():
            raise RuntimeError(f"missing validation log: {log_path}")
        exit_path = require_exit(run, expected)
        content = log_path.read_bytes()
        out = ATTEMPTS_OUT / f"{label}.log.gz"
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_bytes(gzip_bytes(content))
        exit_out = ATTEMPTS_OUT / f"{label}.exit"
        shutil.copyfile(exit_path, exit_out)
        attempts[label] = {
            "log": str(out.relative_to(REVIEW)),
            "log_sha256": sha256(content),
            "exit": expected,
        }
    return attempts


def record_hashes() -> dict[str, object]:
    base = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    product_patch = subprocess.check_output(
        ["git", "diff", base, "--", "apps"], cwd=ROOT
    )
    test_patch = subprocess.check_output(
        ["git", "diff", base, "--", "tests"], cwd=ROOT
    )
    product_files = subprocess.check_output(
        ["git", "diff", "--name-only", base, "--", "apps"], cwd=ROOT, text=True
    ).splitlines()
    test_files = subprocess.check_output(
        ["git", "diff", "--name-only", base, "--", "tests"], cwd=ROOT, text=True
    ).splitlines()

    def hash_files(paths: list[str]) -> dict[str, str]:
        return {path: sha256((ROOT / path).read_bytes()) for path in paths}

    binaries = [
        "build/xuzz",
        "build/gleditor",
        "build/gleditor_test",
        "build/xudu-dump",
        "build/libgleditor.so.0",
        "build/zigzag_test",
        "build/xuzz_test",
        "build/xudu_test",
    ]
    frozen = {
        name: sha256((STAGING / "frozen" / name).read_bytes())
        for name in ("libgleditor.so.0", "xudu-dump")
        if (STAGING / "frozen" / name).is_file()
    }
    current_engine = {
        "build/libgleditor.so.0": sha256((ROOT / "build/libgleditor.so.0").read_bytes()),
        "build/xudu-dump": sha256((ROOT / "build/xudu-dump").read_bytes()),
    }
    return {
        "base_commit": base,
        "product_patch_sha256": sha256(product_patch),
        "test_patch_sha256": sha256(test_patch),
        "product_source_sha256": hash_files(product_files),
        "test_source_sha256": hash_files(test_files),
        "binary_sha256": hash_files([path for path in binaries if (ROOT / path).is_file()]),
        "frozen_final_hashes_file_sha256": sha256(
            (STAGING / "frozen-hashes.txt").read_bytes()
        ),
        "frozen_final_engine_binaries": frozen,
        "current_engine_binaries": current_engine,
        "engine_binary_matches_frozen_final": {
            name: frozen.get(name.removeprefix("build/")) == digest
            for name, digest in current_engine.items()
        },
    }


def archive_strict() -> dict[str, object]:
    if not STRICT.is_dir():
        raise RuntimeError(f"missing strict-core staging directory: {STRICT}")
    frame_data: list[dict[str, object]] = []
    for ppm in sorted(STRICT.glob("*.ppm")):
        width, height, rgb = read_ppm(ppm)
        target = STRICT_OUT / (ppm.stem + ".png")
        target.parent.mkdir(parents=True, exist_ok=True)
        write_png(target, width, height, rgb)
        verify_png(target, width, height, rgb)
        raw_target = STRICT_OUT / (ppm.name + ".gz")
        raw_target.write_bytes(gzip_bytes(ppm.read_bytes()))
        if gzip.decompress(raw_target.read_bytes()) != ppm.read_bytes():
            raise RuntimeError("compressed PPM differs from source")
        frame_data.append(
            {
                "name": ppm.stem,
                "ppm_file": raw_target.name,
                "ppm_storage": "byte-exact deterministic gzip",
                "width": width,
                "height": height,
                "ppm_sha256": sha256(ppm.read_bytes()),
                "rgb_sha256": sha256(rgb),
                "png_sha256": sha256(target.read_bytes()),
                "png_rgb_equal_to_ppm": True,
            }
        )
    if not frame_data:
        raise RuntimeError("strict-core staging has no PPM frames")
    (STRICT_OUT / "frame-provenance.json").write_text(
        json.dumps(frame_data, indent=2) + "\n", encoding="utf-8"
    )
    for source in sorted(STRICT.iterdir()):
        if source.is_file() and source.suffix != ".ppm":
            shutil.copyfile(source, STRICT_OUT / source.name)

    comparison_source = Path("/tmp/spatial-eighteenth-strict.log")
    if not comparison_source.is_file():
        raise RuntimeError(f"missing strict comparison log: {comparison_source}")
    comparison_bytes = comparison_source.read_bytes()
    comparison = comparison_bytes.decode("utf-8", errors="replace")
    (STRICT_OUT / "comparison.log").write_bytes(comparison_bytes)
    required = (
        "0/480000 pixels differ",
        "2450/480000 pixels differ (0.5104%)",
        "2446/480000 pixels differ (0.5096%)",
        "too few page draws to split",
    )
    for marker in required:
        if marker not in comparison:
            raise RuntimeError(f"strict comparison lacks required result: {marker}")
    return {
        "exit": 0,
        "backends": ["OpenGL", "OpenGL ES", "Vulkan"],
        "GL_GLES_ordinary_differing_pixels": 0,
        "Vulkan_ordinary_percent": 0.5104,
        "Vulkan_overlay_percent": 0.5096,
        "actual_threaded_split_exercised": False,
        "frames": len(frame_data),
        "comparison_log_sha256": sha256(comparison_bytes),
    }


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--native-complete",
        action="store_true",
        help="confirm the owning native evidence workflow is complete",
    )
    args = parser.parse_args()
    if not args.native_complete:
        raise SystemExit("refusing to archive before native completion; pass --native-complete when ready")
    if not (REVIEW / "native").is_dir():
        raise RuntimeError("native-package evidence must be archived before this helper runs")

    attempts = archive_logs()
    strict = archive_strict()
    provenance = {
        "source_revision": subprocess.check_output(
            ["git", "rev-parse", "HEAD"], cwd=ROOT, text=True
        ).strip(),
        "strict_core": strict,
        "attempts": attempts,
        "hashes": record_hashes(),
        "test_results": {
            "final_ui": {"passed": 246, "total": 246, "exit": 0},
            "fresh_xuzz": {"passed": 107, "total": 107, "exit": 0},
            "owning_publication_packages": {"passed": 70, "total": 70, "exit": 0},
            "library": "Reused default library 799/799 and mip/cache 53/53 only after byte-identical library/test hash checks; not new runs or A11y-variant evidence",
        },
        "excluded": [
            "private GPG keys and profiles",
            "XDG profiles and caches",
            "permascroll or store binaries",
            "application and test binaries",
            "downloaded package payloads",
        ],
    }
    (STRICT_OUT / "validation-provenance.json").write_text(
        json.dumps(provenance, indent=2) + "\n", encoding="utf-8"
    )
    print(f"Archived {strict['frames']} verified strict-core PNGs and validation logs")
    print("No native evidence or binary payloads were copied by this helper")


if __name__ == "__main__":
    main()
