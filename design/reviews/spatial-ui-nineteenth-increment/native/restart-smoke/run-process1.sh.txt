#!/usr/bin/env bash
set -euo pipefail
repo=$PWD
run=$repo/build/validation-nineteenth/native/restart-smoke
app=$repo/build/validation-nineteenth/frozen/xuzz
libdir=$repo/build/validation-nineteenth/frozen
export SDL_VIDEODRIVER=x11 SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg PATH=/tmp/spatial-xvfb/usr/bin:$PATH
args=(--backend opengl --resize 1440,1000 --profile --wait-ms 5000
  --chord Ctrl+O --wait-ms 800 --dump-a11y
  --capture "$run/raw/01-download-list.ppm"
  --key right --key right --key right --key right --key right
  --key tab --wait-ms 100 --key enter --wait-ms 800 --dump-a11y
  --capture "$run/raw/02-failed-form.ppm"
  --wait-ms 300 --key escape --chord Ctrl+Q)
printf '%q ' "$app" "${args[@]}" > "$run/argv-process1.txt"
printf '\n' >> "$run/argv-process1.txt"
env | sort > "$run/environment.txt"
set +e
LD_LIBRARY_PATH="$libdir${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" timeout --signal=TERM --kill-after=5s 300s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' dbus-run-session -- "$app" "${args[@]}" > "$run/logs/process1.log" 2>&1
code=$?
set -e
printf '%s\n' "$code" > "$run/process1.exit"
exit "$code"
