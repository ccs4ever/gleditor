#!/usr/bin/env bash
set -euo pipefail
run=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-twelfth/candidate-closed
bin=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-twelfth/candidate-bin
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --chord 'Alt+Shift+O' --click-label 'Close Untitled · b' --chord 'Ctrl+Alt+I'
  --dump-a11y --capture "$run/raw/11-identity-closed-page1.ppm" --wait-ms 5000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/12-identity-closed-page2.ppm" --wait-ms 5000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/13-identity-closed-page3.ppm" --wait-ms 5000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/14-identity-closed-page4.ppm" --wait-ms 5000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/15-identity-closed-page5.ppm" --wait-ms 5000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/16-identity-closed-page6.ppm" --wait-ms 5000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/identity-pagewalk-argv.txt"; printf '\n' >> "$run/evidence/identity-pagewalk-argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before-pagewalk.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/identity-pagewalk.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" root="$run/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$root" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 if [[ -e "$run/data/xudu/activity" ]]; then "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true; else printf absent > "$target/activity.ops.txt"; fi
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 11-identity-closed-page1 12-identity-closed-page2 13-identity-closed-page3 14-identity-closed-page4 15-identity-closed-page5 16-identity-closed-page6; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/identity-pagewalk-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-pagewalk.txt"
