#!/usr/bin/env bash
set -euo pipefail
run=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-twelfth/candidate-closed
bin=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-twelfth/candidate-bin
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
passage=$'Harbor left passage.\nMeadow second passage.\nRight one.\nRight two.\nRight three.\n'
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --chord Ctrl+N --type "$passage" --select 0,20 --chord Ctrl+T --chord Ctrl+2
  --select 0,20 --chord 'Ctrl+Alt+[' --select 21,43 --chord 'Ctrl+Alt+['
  --select 44,54 --chord 'Ctrl+Alt+]' --select 55,65 --chord 'Ctrl+Alt+]' --select 66,78 --chord 'Ctrl+Alt+]'
  --chord 'Ctrl+Alt+L' --select 0,20 --chord 'Alt+Shift+N'
  --chord 'Alt+Shift+J' --chord 'Alt+Shift+L' --chord 'Alt+Shift+L'
  --chord 'Alt+Shift+X' --chord 'Alt+Shift+J' --chord 'Alt+Shift+J' --chord 'Alt+Shift+X'
  --dump-a11y --capture "$run/raw/00-selected.ppm" --wait-ms 5000
  --click-label 'Inspect members' --dump-a11y --capture "$run/raw/01-inspected.ppm" --wait-ms 5000
  --chord 'Alt+Shift+Return' --dump-a11y --capture "$run/raw/02-enter-b1.ppm" --wait-ms 5000
  --chord 'Alt+Shift+O' --dump-a11y --capture "$run/raw/03-origin.ppm" --wait-ms 5000
  --click-label 'Close Untitled · b' --dump-a11y --capture "$run/raw/04-closed-before-exit.ppm" --wait-ms 5000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/phase1-argv.txt"; printf '\n' >> "$run/evidence/phase1-argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 240s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/phase1.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" root="$run/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$root" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 if [[ -e "$run/data/xudu/activity" ]]; then "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true; else printf absent > "$target/activity.ops.txt"; fi
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 00-selected 01-inspected 02-enter-b1 03-origin 04-closed-before-exit; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/phase1-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-phase1.txt"
