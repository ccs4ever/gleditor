#!/bin/sh
set -eu
cd "$(dirname "$0")/../.."
run_root=$(mktemp -d "$PWD/build/validation-twenty-first/xdg-$1.XXXXXX")
mkdir -p "$run_root/gpg" "$run_root/tmp"
chmod 700 "$run_root/gpg"
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME="$run_root/data" XDG_CONFIG_HOME="$run_root/config" XDG_CACHE_HOME="$run_root/cache"
export GNUPGHOME="$run_root/gpg" TMPDIR="$run_root/tmp"
case "$1" in
  ui) exec ./build/zigzag_test ;;
  xuzz) exec ./build/xuzz_test ;;
  engine) exec ./build/xudu_test --gtest_filter='-SwarmTest.*:MutableNameTest.*:PublicationOutboxNetworkTest.*:E2EBinaryOrchestrationTest.*:AnimationTransclusionTest.*:BeamProgressiveLoadingTest.*' --gtest_output="xml:$PWD/build/validation-twenty-first/engine-tests.xml" ;;
  *) exit 2 ;;
esac
