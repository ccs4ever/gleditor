#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fifteenth
run=$PWD/build/validation-sixteenth/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$base/frozen-final/xuzz" --backend opengl --resize 1440,1000 --profile
 --wait-ms 5000 --chord 'Ctrl+Alt+Shift+L' --key right --key enter --wait-ms 1000
 --dump-a11y --capture "$run/raw/07-downloaded-candidate.ppm" --wait-ms 1000
 --key enter --wait-ms 1000 --dump-a11y --capture "$run/raw/08-package-status.ppm" --wait-ms 1000
 --key right --key enter --wait-ms 1000 --dump-a11y --capture "$run/raw/09-package-review.ppm" --wait-ms 1000
 --key escape --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/package-readiness-argv.txt"; printf '\n' >> "$run/evidence/package-readiness-argv.txt"
LD_LIBRARY_PATH="$base/frozen-final${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/package-readiness.log" 2>&1
