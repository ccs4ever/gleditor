#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fifteenth
run=$PWD/build/validation-sixteenth/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$base/frozen-final/xuzz" --backend opengl --resize 1440,1000 --profile
 --chord 'Ctrl+Alt+Shift+P' --dump-a11y --capture "$run/raw/03-prepare-package-form.ppm" --wait-ms 1500
 --key tab --key tab --type 'Spatial review' --key tab --type 'curations:links' --key enter --wait-ms 4000
 --dump-a11y --capture "$run/raw/04-real-package-status.ppm" --wait-ms 1500
 --key escape --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/prepare-package-argv.txt"; printf '\n' >> "$run/evidence/prepare-package-argv.txt"
LD_LIBRARY_PATH="$base/frozen-final${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/prepare-package.log" 2>&1
