#!/usr/bin/env bash
set -euo pipefail
base=$PWD/build/validation-fifteenth
run=$PWD/build/validation-sixteenth/native
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
passage=$'Harbor left passage.\nMeadow second passage.\nRight one.\nRight two.\nRight three.\n'
app=("$base/frozen-final/xuzz" --backend opengl --resize 1440,1000 --profile
 --chord Ctrl+N --type "$passage" --select 0,20 --chord Ctrl+T --chord Ctrl+2
 --select 0,20 --chord 'Ctrl+Alt+[' --select 21,43 --chord 'Ctrl+Alt+['
 --select 44,54 --chord 'Ctrl+Alt+]' --select 55,65 --chord 'Ctrl+Alt+]' --select 66,78 --chord 'Ctrl+Alt+]'
 --chord 'Ctrl+Alt+L' --select 0,20 --chord 'Alt+Shift+N'
 --dump-a11y --capture "$run/raw/00-authored-link.ppm" --wait-ms 1500
 --chord 'Ctrl+Shift+S' --dump-a11y --capture "$run/raw/01-local-publication-form.ppm" --wait-ms 1500
 --key escape --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/publication-form-argv.txt"; printf '\n' >> "$run/evidence/publication-form-argv.txt"
LD_LIBRARY_PATH="$base/frozen-final${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL=xuzz.keymap=debug timeout --signal=TERM --kill-after=5s 180s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/publication-form.log" 2>&1
