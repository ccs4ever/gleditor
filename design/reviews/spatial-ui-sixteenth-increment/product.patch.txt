+      }
+      if (!editions.empty())

+        choices.optionValues = {""};
+      }
+      const auto describe = [&](const InfoHash &hash) {
+        const auto *package =
+            reader.verifiedPackage(ReaderLinkPackages::keyOf(hash, 0));
+        const auto preference = reader.preference(hash);
+        std::size_t left = 0, right = 0;
+        for (const auto &link : package->links) {
+          left += link.left.size();
+          right += link.right.size();
+        }

+               (preference.emphasis == PackageEmphasis::Quiet ? "quiet"

+               std::to_string(left) + "/R" + std::to_string(right);

+      auto control = actions({"Refresh comparison", "Keep current drawing",

+ReaderLinkPackages::editions(const InfoHash &candidate) const {
+  const auto found = verified.find(candidate);
+  if (found == verified.end()) return {};
+  std::vector<InfoHash> result;
+  for (const auto &[hash, package] : verified)
+    if (hash != candidate && preference(hash).retained &&
+        package.curator == found->second.curator &&
+        package.salt == found->second.salt)
+      result.push_back(hash);
+  std::ranges::sort(result, [&](const auto &a, const auto &b) {
+    const auto left = verified.at(a).sequence, right = verified.at(b).sequence;
+    return left != right ? left > right : a < b;
+  });
+  return result;
+}
