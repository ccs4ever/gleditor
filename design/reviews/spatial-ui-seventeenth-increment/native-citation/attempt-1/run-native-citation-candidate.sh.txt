#!/usr/bin/env bash
set -euo pipefail

# Prepared only: do not run until the parent freezes the canonical candidate.
# Args: frozen xuzz executable, frozen library directory, frozen xudu-dump.
repo=$PWD
base=$repo/build/validation-seventeenth
run=$base/native-candidate
baseline=$base/native-baseline
app=${1:?pass the frozen candidate xuzz path}
libdir=${2:?pass the frozen candidate library directory}
dumper=${3:?pass the frozen candidate xudu-dump path}

[[ -x $app && -x $dumper && -d $libdir ]] || {
  printf 'Candidate binary, library directory, or dumper is missing\n' >&2
  exit 2
}
if [[ -e $run ]]; then
  printf 'Refusing to overwrite existing candidate run: %s\n' "$run" >&2
  exit 2
fi
mkdir -p "$run"/{data,config,cache,gpg,raw,evidence}
cp -a "$baseline/data/." "$run/data/"
cp -a "$baseline/config/." "$run/config/"
chmod 700 "$run/gpg"
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export GNUPGHOME=$run/gpg
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH

printf '%s\n' "$app" > "$run/evidence/binary-path.txt"
sha256sum "$app" "$dumper" > "$run/evidence/binary-hashes.txt"
printf '%s\n' "SDL_VIDEODRIVER=$SDL_VIDEODRIVER" "SDL_AUDIODRIVER=$SDL_AUDIODRIVER" \
  "LIBGL_ALWAYS_SOFTWARE=$LIBGL_ALWAYS_SOFTWARE" "XDG_DATA_HOME=$XDG_DATA_HOME" \
  "XDG_CONFIG_HOME=$XDG_CONFIG_HOME" "XDG_CACHE_HOME=$XDG_CACHE_HOME" \
  "GNUPGHOME=$GNUPGHOME" > "$run/evidence/environment.txt"

snapshot() {
  local tag=$1 dest=$run/evidence/checkpoints/$1
  mkdir -p "$dest"
  mapfile -t stores < <(find "$run/data/xudu/xanadocs" -mindepth 1 -maxdepth 1 -type d ! -name default | sort)
  for i in "${!stores[@]}"; do
    "$dumper" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$dest/source-$i.ops.txt"
    sha256sum "${stores[$i]}/ops.nodes" "${stores[$i]}/store.tables" > "$dest/source-$i.files.sha256"
  done
  "$dumper" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$dest/activity.ops.txt"
  if [[ -f "$run/data/xudu/package-visibility" ]]; then
    cp "$run/data/xudu/package-visibility" "$dest/package-visibility.bencode"
    sha256sum "$run/data/xudu/package-visibility" > "$dest/package-visibility.sha256"
  fi
}

# The baseline-authored package and exact route are reused. The final Right
# highlights Open related publication; the first capture records review before
# opening. Further actions are explicit and captured as separate states.
ui=(--backend opengl --resize 1440,1000 --profile --wait-ms 5000
  --chord Ctrl+Alt+Shift+L --key right --key enter --wait-ms 500
  --key tab --key right --chord Shift+Tab --key enter --wait-ms 500
  --key right --key enter --wait-ms 500 --key right)

snapshot before
capture_review=(--dump-a11y --capture "$run/raw/00-review-before-open.ppm")
app_cmd=("$app")
app_args=("${ui[@]}" "${capture_review[@]}" --wait-ms 300 --key enter
  --wait-ms 700 --dump-a11y --capture "$run/raw/01-open-immediate.ppm"
  --wait-ms 1500 --dump-a11y --capture "$run/raw/02-open-settled.ppm"
  --key right --key enter --wait-ms 500 --dump-a11y
  --capture "$run/raw/03-explicit-retry-immediate.ppm" --wait-ms 1500 --key enter
  --wait-ms 300 --dump-a11y --capture "$run/raw/04-retry-settled.ppm"
  --key escape --wait-ms 300 --chord Ctrl+Alt+Shift+L --key right --key enter
  --wait-ms 500 --key tab --key right --chord Shift+Tab --key enter
  --wait-ms 500 --key right --key enter --wait-ms 500 --key right --key enter
  --wait-ms 800 --dump-a11y --capture "$run/raw/05-same-citation-again.ppm"
  --key escape --chord Ctrl+Q)
printf '%q ' "${app_cmd[@]}" "${app_args[@]}" > "$run/evidence/citation-candidate-argv.txt"
printf '\n' >> "$run/evidence/citation-candidate-argv.txt"
run_exit=0
if LD_LIBRARY_PATH="$libdir${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" \
  timeout --signal=TERM --kill-after=5s 300s xvfb-run -a \
    -s '-noreset -screen 0 1600x1200x24' \
    "${app_cmd[@]}" "${app_args[@]}" > "$run/evidence/citation-candidate.log" 2>&1; then
  run_exit=0
else
  run_exit=$?
fi
printf '%s\n' "$run_exit" > "$run/evidence/citation-candidate.exit"
snapshot after
sha256sum "$run/raw/"*.ppm > "$run/evidence/raw-ppm-hashes.txt"
exit "$run_exit"
