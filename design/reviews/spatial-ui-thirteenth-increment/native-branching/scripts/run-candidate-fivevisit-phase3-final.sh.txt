#!/usr/bin/env bash
set -euo pipefail
root=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-thirteenth
profile=$root/candidate-fivevisit
run=$profile/final-phase3; bin=$root/candidate-build5
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$profile/data XDG_CONFIG_HOME=$profile/config XDG_CACHE_HOME=$profile/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --dump-a11y --capture "$run/raw/00-after-restart-b1-closed.ppm" --wait-ms 5000
  --click-label 'Inspect members' --dump-a11y --capture "$run/raw/01-inspector-page1.ppm" --wait-ms 2000
  --click-label 'Next controls page' --dump-a11y --capture "$run/raw/02-inspector-page2.ppm" --wait-ms 2000
  --chord 'Alt+Shift+Return' --dump-a11y --capture "$run/raw/03-enter-refused.ppm" --wait-ms 3000
  --chord 'Alt+Shift+Escape' --dump-a11y --capture "$run/raw/04-cancelled.ppm" --wait-ms 3000
  --chord Ctrl+H --dump-a11y --capture "$run/raw/05-hypertime-map.ppm" --wait-ms 3000
  --click-label 'T · Version b1' --wait-ms 5000 --chord Ctrl+H --dump-a11y --capture "$run/raw/06-reopened-b1.ppm" --wait-ms 3000
  --chord 'Alt+Shift+Return' --dump-a11y --capture "$run/raw/07-retry-enter-b1.ppm" --wait-ms 5000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/argv.txt"; printf '\n' >> "$run/evidence/argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 240s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/session.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" rootstores="$profile/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$rootstores" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$profile/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 "$bin/xudu-dump" --section=ops --permascroll="$profile/data/xuzz/permascroll/default" "$profile/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 00-after-restart-b1-closed 01-inspector-page1 02-inspector-page2 03-enter-refused 04-cancelled 05-hypertime-map 06-reopened-b1 07-retry-enter-b1; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after.txt"
