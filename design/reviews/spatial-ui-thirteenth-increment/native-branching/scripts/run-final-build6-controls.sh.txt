#!/usr/bin/env bash
set -euo pipefail
root=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-thirteenth
profile=$root/candidate-fivevisit
run=$profile/final-phase4
bin=$root/candidate-build6
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$profile/data XDG_CONFIG_HOME=$profile/config XDG_CACHE_HOME=$profile/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --dump-a11y --capture "$run/raw/00-current6-after-retry.ppm" --wait-ms 4000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/01-walks-current6.ppm" --wait-ms 2500
  --key enter --key up --key enter --key tab --key tab --key enter
  --dump-a11y --capture "$run/raw/02-preview5-current6.ppm" --wait-ms 3000
  --key escape --dump-a11y --capture "$run/raw/03-current6-after-preview.ppm" --wait-ms 1500
  --chord 'Alt+Shift+B' --dump-a11y --capture "$run/raw/04-back-from6.ppm" --wait-ms 3500
  --chord 'Alt+Shift+Y' --dump-a11y --capture "$run/raw/05-forward-to6.ppm" --wait-ms 3500
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/argv-controls.txt"; printf '\n' >> "$run/evidence/argv-controls.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before-controls.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 240s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/session-controls.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" rootstores="$profile/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$rootstores" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$profile/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 "$bin/xudu-dump" --section=ops --permascroll="$profile/data/xuzz/permascroll/default" "$profile/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 00-current6-after-retry 01-walks-current6 02-preview5-current6 03-current6-after-preview 04-back-from6 05-forward-to6; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/controls-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-controls.txt"
