#!/usr/bin/env bash
set -euo pipefail
root=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-thirteenth
run=$root/native; bin=$root/frozen
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --dump-a11y --capture "$run/raw/07-restart-current.ppm" --wait-ms 5000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/08-walks-before-preview.ppm" --wait-ms 5000
  --click-label 'Branch visit 2 from 1: untitled-20261015-203313 · page 1/1'
  --dump-a11y --capture "$run/raw/09-walks-selected-visit2.ppm" --wait-ms 2000
  --key tab --key tab --key enter --dump-a11y --capture "$run/raw/10-walks-preview-visit2.ppm" --wait-ms 5000
  --key escape --dump-a11y --capture "$run/raw/11-current-after-preview.ppm" --wait-ms 3000
  --chord 'Alt+Shift+Y' --dump-a11y --capture "$run/raw/12-forward-choices-root.ppm" --wait-ms 3000
  --click-label 'Visit 2: untitled-20261015-203313 · page 1/1' --key enter
  --dump-a11y --capture "$run/raw/13-forward-visit2.ppm" --wait-ms 5000
  --chord 'Alt+Shift+B' --dump-a11y --capture "$run/raw/14-back-root-after-visit2.ppm" --wait-ms 5000
  --chord 'Alt+Shift+Y' --dump-a11y --capture "$run/raw/15-forward-choices-again.ppm" --wait-ms 3000
  --click-label 'Visit 3: untitled-20261015-203313 · page 1/1' --key enter
  --dump-a11y --capture "$run/raw/16-forward-visit3.ppm" --wait-ms 5000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/17-walks-at-visit3.ppm" --wait-ms 3000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/phase2-argv.txt"; printf '\n' >> "$run/evidence/phase2-argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before-phase2.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 240s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/phase2.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" rootstores="$run/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$rootstores" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 if [[ -e "$run/data/xudu/activity" ]]; then "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true; else printf absent > "$target/activity.ops.txt"; fi
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 07-restart-current 08-walks-before-preview 09-walks-selected-visit2 10-walks-preview-visit2 11-current-after-preview 12-forward-choices-root 13-forward-visit2 14-back-root-after-visit2 15-forward-choices-again 16-forward-visit3 17-walks-at-visit3; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/phase2-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-phase2.txt"
