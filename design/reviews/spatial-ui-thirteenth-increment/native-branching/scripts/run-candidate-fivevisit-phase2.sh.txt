#!/usr/bin/env bash
set -euo pipefail
root=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-thirteenth
run=$root/candidate-fivevisit; bin=$root/candidate-final
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --dump-a11y --capture "$run/raw/15-restart-current3.ppm" --wait-ms 5000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/16-restart-walks-current3.ppm" --wait-ms 3000
  --key enter --key down --key down --key enter --key tab --key tab --key enter
  --dump-a11y --capture "$run/raw/17-restart-preview5-current3.ppm" --wait-ms 4000
  --key escape --dump-a11y --capture "$run/raw/18-current3-after-restart-preview.ppm" --wait-ms 2000
  --chord 'Alt+Shift+Y' --dump-a11y --capture "$run/raw/19-restart-no-forward-at-leaf3.ppm" --wait-ms 2000
  --chord 'Alt+Shift+B' --dump-a11y --capture "$run/raw/20-restart-back-visit2.ppm" --wait-ms 3000
  --chord 'Alt+Shift+Y' --key space --key down --key enter --key enter
  --dump-a11y --capture "$run/raw/21-restart-forward-sibling4.ppm" --wait-ms 4000
  --chord 'Alt+Shift+B' --chord 'Alt+Shift+Y' --key enter
  --dump-a11y --capture "$run/raw/22-restart-forward-visit3.ppm" --wait-ms 4000
  --chord 'Alt+Shift+X' --dump-a11y --capture "$run/raw/23-left-pin-active-before-close.ppm" --wait-ms 2000
  --click-label 'Close Untitled · b' --dump-a11y --capture "$run/raw/24-b1-closed.ppm" --wait-ms 4000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/phase2-argv.txt"; printf '\n' >> "$run/evidence/phase2-argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before-phase2.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 300s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/phase2.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" rootstores="$run/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$rootstores" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 15-restart-current3 16-restart-walks-current3 17-restart-preview5-current3 18-current3-after-restart-preview 19-restart-no-forward-at-leaf3 20-restart-back-visit2 21-restart-forward-sibling4 22-restart-forward-visit3 23-left-pin-active-before-close 24-b1-closed; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/phase2-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-phase2.txt"
