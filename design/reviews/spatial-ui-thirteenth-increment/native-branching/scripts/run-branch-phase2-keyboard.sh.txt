#!/usr/bin/env bash
set -euo pipefail
root=/data/git/gleditor-restore/.worktrees/spatial-ui-overhaul/build/validation-thirteenth
run=$root/native; bin=$root/frozen
export SDL_VIDEODRIVER=offscreen SDL_AUDIODRIVER=dummy LIBGL_ALWAYS_SOFTWARE=1
export XDG_DATA_HOME=$run/data XDG_CONFIG_HOME=$run/config XDG_CACHE_HOME=$run/cache
export PATH=/tmp/spatial-xvfb/usr/bin:$PATH
app=("$bin/xuzz" --backend opengl --resize 1440,1000 --profile
  --dump-a11y --capture "$run/raw/18-restart-root.ppm" --wait-ms 5000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/19-walks-root-before-preview.ppm" --wait-ms 3000
  --key enter --key down --key enter --key tab --key tab --key enter
  --dump-a11y --capture "$run/raw/20-walks-preview-visit2.ppm" --wait-ms 5000
  --key escape --dump-a11y --capture "$run/raw/21-current-after-preview.ppm" --wait-ms 3000
  --chord 'Alt+Shift+Y' --dump-a11y --capture "$run/raw/22-forward-choices-root.ppm" --wait-ms 3000
  --key enter --dump-a11y --capture "$run/raw/23-forward-visit2.ppm" --wait-ms 5000
  --chord 'Alt+Shift+B' --dump-a11y --capture "$run/raw/24-back-root-after-visit2.ppm" --wait-ms 5000
  --chord 'Alt+Shift+Y' --key space --key down --key enter --key enter
  --dump-a11y --capture "$run/raw/25-forward-visit3.ppm" --wait-ms 5000
  --chord 'Ctrl+Alt+W' --dump-a11y --capture "$run/raw/26-walks-at-visit3.ppm" --wait-ms 3000
  --chord Ctrl+Q)
printf '%q ' "${app[@]}" > "$run/evidence/phase2-keyboard-argv.txt"; printf '\n' >> "$run/evidence/phase2-keyboard-argv.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-before-phase2-keyboard.txt"
LD_LIBRARY_PATH="$bin${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}" SPDLOG_LEVEL='xuzz.keymap=debug' timeout --signal=TERM --kill-after=5s 240s xvfb-run -a -s '-noreset -screen 0 1600x1200x24' "${app[@]}" > "$run/evidence/phase2-keyboard.log" 2>&1 & pid=$!
snapshot() {
 local step=$1 target="$run/evidence/checkpoints/$1" rootstores="$run/data/xudu/xanadocs"
 mkdir -p "$target"; mapfile -t stores < <(find "$rootstores" -mindepth 1 -maxdepth 1 -type d ! -name default 2>/dev/null | sort)
 { printf 'count=%s\n' "${#stores[@]}"; for i in "${!stores[@]}"; do printf '%s=%s\n' "$i" "${stores[$i]}"; "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "${stores[$i]}" > "$target/source-$i.ops.txt" 2>&1 || true; done; } > "$target/profile-stores.txt"
 if [[ -e "$run/data/xudu/activity" ]]; then "$bin/xudu-dump" --section=ops --permascroll="$run/data/xuzz/permascroll/default" "$run/data/xudu/activity" > "$target/activity.ops.txt" 2>&1 || true; else printf absent > "$target/activity.ops.txt"; fi
 sha256sum "$target"/*.txt > "$target/SHA256SUMS"
}
for step in 18-restart-root 19-walks-root-before-preview 20-walks-preview-visit2 21-current-after-preview 22-forward-choices-root 23-forward-visit2 24-back-root-after-visit2 25-forward-visit3 26-walks-at-visit3; do
 f="$run/raw/$step.ppm"; ready=false
 for _ in $(seq 1 120); do [[ -s "$f" && $(stat -c %s "$f") -eq 4320017 ]] && { ready=true; break; }; kill -0 "$pid" 2>/dev/null || break; sleep .5; done
 if [[ $ready == true ]]; then snapshot "$step"; else printf missing-or-incomplete > "$run/evidence/$step.missing"; fi
done
status=0; wait "$pid" || status=$?; printf '%s\n' "$status" > "$run/evidence/phase2-keyboard-exit.txt"
sha256sum "$bin"/{xuzz,libgleditor.so.0,xudu-dump} > "$run/evidence/binary-hashes-after-phase2-keyboard.txt"
